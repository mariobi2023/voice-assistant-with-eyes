# Backend dependencies (Python 3.12). Install inside the virtual environment, IN THIS ORDER:
#   1) CPU-only PyTorch (no CUDA, ~115 MB):
#        pip install torch==2.10.0 torchvision==0.25.0 --index-url https://download.pytorch.org/whl/cpu
#   2) CLIP (used once by YOLO-World to read our class names; not on PyPI):
#        pip install git+https://github.com/ultralytics/CLIP.git
#      If git reports an SSL certificate error on Windows:
#        git -c http.sslBackend=schannel clone https://github.com/ultralytics/CLIP.git
#        pip install ./CLIP
#   3) Everything else:
#        pip install -r requirements.txt

# --- Video and images ---
# Ultralytics needs "opencv-python" and EasyOCR needs "opencv-python-headless". Both install the
# same "cv2" module, so they MUST have the same version or cv2 breaks.
opencv-python==5.0.0.93          # read RTSP, image maths (sharpness, brightness), face detection, JPEG
opencv-python-headless==5.0.0.93
numpy==2.5.5                     # arrays used by OpenCV / PyTorch
Pillow==10.4.0                   # image format needed by imagehash
ImageHash==4.3.1                 # perceptual hash (pHash) to skip near-duplicate frames

# --- AI models (CPU) ---
torch==2.10.0                    # installed first from the CPU-only index (see step 1 above)
torchvision==0.25.0
easyocr==1.7.2                   # OCR for English, French and Arabic
ultralytics==8.4.215             # YOLO-World object detection with our own class list
sentence-transformers==6.3.0     # multilingual text embeddings (384 dims)

# --- Database ---
psycopg[binary]==3.2.3           # PostgreSQL driver
psycopg-pool==3.2.4              # connection pool shared by the background threads and web requests
pgvector==0.5.1                  # store/read numpy vectors in PostgreSQL "vector" columns

# --- Web API and answer model ---
fastapi==0.141.1                 # the web API (/ask, /status, ...) with automatic /docs page
pydantic==2.14.3                 # request/response shapes of the API
uvicorn[standard]==0.54.0        # the web server that runs FastAPI
google-genai==3.4.0              # Gemini API (answer model)
httpx==0.28.1                    # HTTP client used by google-genai (we turn off connection reuse)
requests==2.34.2                 # HTTP calls to the local Ollama fallback

# --- Settings and network ---
python-dotenv==1.0.1             # load settings and secrets from server/.env
truststore==0.10.5               # use the Windows certificate store for HTTPS (fixes SSL errors)
